// GG09: Scaling i odległość euklidesowa
// Wklej do GeoGebry linia po linii.
// Cel: pokazać, że zmiana skali jednej osi zmienia odległości.

A = (1, 1)
B = (4, 2)
C = (1, 5)
Segment(A, B)
Segment(A, C)

s = Slider(0.2, 5, 0.1, 1)
A2 = (x(A), s * y(A))
B2 = (x(B), s * y(B))
C2 = (x(C), s * y(C))
Segment(A2, B2)
Segment(A2, C2)

dAB = Distance(A, B)
dAC = Distance(A, C)
dAB_scaled = Distance(A2, B2)
dAC_scaled = Distance(A2, C2)

Text("Suwak s zmienia skalę osi y", (0.2, 6.2))
Text("d(A,B) = " + dAB, (0.2, 5.8))
Text("d(A,C) = " + dAC, (0.2, 5.4))
Text("d_s(A,B) = " + dAB_scaled, (0.2, 5.0))
Text("d_s(A,C) = " + dAC_scaled, (0.2, 4.6))
