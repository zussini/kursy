// GG11: Cosine similarity — kierunek wektora zamiast długości
// Przesuwaj B i C. Obserwuj kąty względem wektora A.

O = (0, 0)
A = (2, 1)
B = (4, 2)
C = (1, 3)

Vector(O, A)
Vector(O, B)
Vector(O, C)

angleAB = Angle(A, O, B)
angleAC = Angle(A, O, C)
cosAB = cos(angleAB)
cosAC = cos(angleAC)

Text("cos(A,B) = " + cosAB, (0.2, 3.8))
Text("cos(A,C) = " + cosAC, (0.2, 3.4))
Text("B = 2A, więc kierunek ten sam i cos(A,B)=1", (0.2, 3.0))
