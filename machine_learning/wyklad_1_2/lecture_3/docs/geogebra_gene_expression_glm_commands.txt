# Średnie
overall = (1.7 + 2.1 + 2.3 + 2.7 + 3.1 + 3.4 + 3.7 + 4.2) / 8
controlMean = (1.7 + 2.1 + 2.3 + 2.7) / 4
mutantMean = (3.1 + 3.4 + 3.7 + 4.2) / 4

# Linie średnich
meanLine: y = overall

# SSE modelu zerowego
SSE_mean = (1.7-overall)^2 + (2.1-overall)^2 + (2.3-overall)^2 + (2.7-overall)^2 + (3.1-overall)^2 + (3.4-overall)^2 + (3.7-overall)^2 + (4.2-overall)^2

# SSE modelu grupowego
SSE_fit = (1.7-controlMean)^2 + (2.1-controlMean)^2 + (2.3-controlMean)^2 + (2.7-controlMean)^2 + (3.1-mutantMean)^2 + (3.4-mutantMean)^2 + (3.7-mutantMean)^2 + (4.2-mutantMean)^2

# Test F dla dwóch grup
F = ((SSE_mean - SSE_fit) / (2 - 1)) / (SSE_fit / (8 - 2))
