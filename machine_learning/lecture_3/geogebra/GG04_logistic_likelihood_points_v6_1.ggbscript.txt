# GG04 likelihood w regresji logistycznej dla kilku punktów
beta0 = -4
beta1 = 0.8
p(x) = 1 / (1 + exp(-(beta0 + beta1*x)))
x1 = 5
y1 = 0
x2 = 8
y2 = 0
x3 = 11
y3 = 1
x4 = 14
y4 = 1
p1 = p(x1)
p2 = p(x2)
p3 = p(x3)
p4 = p(x4)
logL = y1*ln(p1)+(1-y1)*ln(1-p1)+y2*ln(p2)+(1-y2)*ln(1-p2)+y3*ln(p3)+(1-y3)*ln(1-p3)+y4*ln(p4)+(1-y4)*ln(1-p4)
P1 = (x1, y1)
P2 = (x2, y2)
P3 = (x3, y3)
P4 = (x4, y4)
Text("log-likelihood = " + Round(logL, 3), (4, 1.1))
