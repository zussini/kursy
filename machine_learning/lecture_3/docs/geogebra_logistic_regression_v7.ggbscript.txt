// Demo 2: probability, odds, log-odds
prob = Slider(0.01, 0.99, 0.01)
odds = prob / (1 - prob)
logodds = ln(odds)
A_odds = (prob, odds)
B_logodds = (prob, logodds)
odds_fun(x) = x / (1 - x)
logit_fun(x) = ln(x / (1 - x))
Function(odds_fun, 0.01, 0.99)
Function(logit_fun, 0.01, 0.99)
